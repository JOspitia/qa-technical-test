Solicitud de autenticación (login)

REQUEST:
    -Operation: auth login
    -Endpoint: https://dummyjson.com/auth/login
    -Method: POST
    -Content-Type: application/json
    -Body:
        {
            "username": "{{username}}",
            "password": "{{password}}"
        }

RESPONSE:
    -code: 200
    -body:
        {
            "accessToken": "{{accessToken}}",
            "refreshToken": "{{refreshToken}}",
            "id": "{{id}}",
            "username": "{{username}}",
            "email": "{{email}}",
            "firstName": "{{firstName}}",
            "lastName": "{{lastName}}",
            "gender": "{{gender}}",
            "image": "{{image}}"
        }
    
Solicitud de autenticación con credenciales invalidas

    -Operation: auth login
    -Endpoint: https://dummyjson.com/auth/login
    -Method: POST
    -Content-Type: application/json
    -Body:
        {
            "username": "{{username}}",
            "password": "{{password}}"
        }

RESPONSE:
    -code: 400
    -body: 
        {
            "message": "Invalid credentials"
        }
